import React from "react";
import { BrowserRouter, Routes, Route, Navigate } from "react-router-dom";
import Login from "./pages/auth/Login";
import Register from "./pages/auth/Register";
import Home from "./pages/auth/Home";

// Component bảo vệ route
const PrivateRoute = ({ children }) => {
	const token = localStorage.getItem("token");
	return token ? children : <Navigate to="/login" />;
};

function App() {
	return (
		<BrowserRouter>
			<Routes>
				<Route path="/login" element={<Login />} />
				<Route path="/register" element={<Register />} />
				<Route
					path="/home"
					element={
						<PrivateRoute>
							<Home />
						</PrivateRoute>
					}
				/>
				<Route
					path="/"
					element={
						<PrivateRoute>
							<Home />
						</PrivateRoute>
					}
				/>
			</Routes>
		</BrowserRouter>
	);
}

export default App;

import axios from "axios";

const API_URL = "http://localhost:5000/api/auth";

const authService = {
	// 1. Đăng ký
	register: async (name, email, password) => {
		try {
			const response = await axios.post(`${API_URL}/register`, {
				name,
				email,
				password,
			});
			console.log("Register response:", response.data);
			return response.data;
		} catch (error) {
			console.error("Register error:", error);
			throw error;
		}
	},

	// 2. Đăng nhập
	login: async (email, password) => {
		try {
			const response = await axios.post(`${API_URL}/login`, {
				email,
				password,
			});

			console.log("Login response:", response.data);

			// ✅ SỬA: Backend trả về token trong data.token
			if (response.data.success && response.data.data?.token) {
				// Lưu token
				localStorage.setItem("token", response.data.data.token);
				// ✅ THÊM: Lưu thông tin user
				localStorage.setItem("user", JSON.stringify(response.data.data.user));
			}

			return response.data;
		} catch (error) {
			console.error("Login error:", error);
			throw error;
		}
	},

	// 3. Lấy thông tin cá nhân
	getProfile: async () => {
		const token = localStorage.getItem("token");

		// ✅ SỬA: Không trả về null, mà throw error hoặc trả về object rõ
		if (!token) {
			return { success: false, message: "Chưa đăng nhập" };
		}

		try {
			const response = await axios.get(`${API_URL}/profile`, {
				headers: { Authorization: `Bearer ${token}` },
			});
			return response.data;
		} catch (error) {
			console.error("GetProfile error:", error);
			throw error;
		}
	},

	// 4. Đăng xuất
	logout: () => {
		localStorage.removeItem("token");
		localStorage.removeItem("user"); // ✅ THÊM: Xóa cả user
		window.location.href = "/login"; // ✅ THÊM: Chuyển về login
	},

	// 5. Lấy thông tin user từ localStorage (dùng cho Header)
	getCurrentUser: () => {
		const userStr = localStorage.getItem("user");
		if (userStr) {
			return JSON.parse(userStr);
		}
		return null;
	},

	// 6. Lấy token
	getToken: () => {
		return localStorage.getItem("token");
	},
};

export default authService;
